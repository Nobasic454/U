local function serialize(v, depth)
    depth = depth or 0
    if depth > 4 then return "..." end
    local t = typeof(v)
    if t == "table" then
        local parts = {}
        for k, val in pairs(v) do
            local key = (type(k) == "string") and k or ("["..tostring(k).."]")
            parts[#parts+1] = key.." = "..serialize(val, depth+1)
        end
        return "{ "..table.concat(parts, ", ").." }"
    elseif t == "EnumItem" then
        return "Enum."..tostring(v.EnumType).."."..v.Name
    elseif t == "Instance" then
        return v.ClassName..'("'..v.Name..'")'
    elseif t == "string" then
        return '"'..v..'"'
    elseif t == "Vector3" then
        return string.format("Vector3(%.2f, %.2f, %.2f)", v.X, v.Y, v.Z)
    elseif t == "CFrame" then
        return string.format("CFrame(%.2f, %.2f, %.2f)", v.X, v.Y, v.Z)
    else
        return tostring(v)
    end
end

local function formatArgs(args)
    local parts = {}
    for i, v in ipairs(args) do
        parts[i] = serialize(v)
    end
    return table.concat(parts, ",\n    ")
end

local function timestamp()
    return string.format("[%05.2f]", tick() % 1000)
end

------------------------------------------------
-- Лог-буфер + print
------------------------------------------------
local logs = {}
local MAX_LOGS = 200

local function log(tag, remote, args, source)
    local entry = {
        tag    = tag,      -- "OUT" | "IN"
        remote = remote,
        source = source or "?",
        args   = args,
        time   = timestamp(),
    }
    table.insert(logs, 1, entry)
    if #logs > MAX_LOGS then table.remove(logs) end

    -- Console
    local goal = (type(args[1]) == "table" and args[1].Goal) and (" | Goal: "..tostring(args[1].Goal)) or ""
    print(string.format(
        "%s [%s] %s  ← %s%s\n    args: {\n    %s\n    }",
        entry.time, tag, remote, source, goal, formatArgs(args)
    ))
end

------------------------------------------------
-- Spy OnClientEvent (входящие от сервера)
-- Мониторим Communicate своего чара + других
------------------------------------------------
local hooked = {}

local function hookCommunicate(character, playerName)
    local remote = character:FindFirstChild("Communicate")
    if not remote or hooked[remote] then return end
    hooked[remote] = true

    remote.OnClientEvent:Connect(function(...)
        local args = {...}
        log("IN  ▼", "Communicate", args, "Server → "..playerName)
    end)
end

-- Себя
local function onCharAdded(char)
    char:WaitForChild("Communicate", 10)
    hookCommunicate(char, LocalPlayer.Name.." (self)")
end
if LocalPlayer.Character then onCharAdded(LocalPlayer.Character) end
LocalPlayer.CharacterAdded:Connect(onCharAdded)

-- Других игроков (видим входящие им от сервера)
for _, p in pairs(Players:GetPlayers()) do
    if p ~= LocalPlayer and p.Character then
        hookCommunicate(p.Character, p.Name)
    end
end
Players.PlayerAdded:Connect(function(p)
    p.CharacterAdded:Connect(function(char)
        char:WaitForChild("Communicate", 10)
        hookCommunicate(char, p.Name)
    end)
end)

do
    local drag, dragStart, startPos = false
    Frame.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 then
            drag = true; dragStart = i.Position; startPos = Frame.Position
        end
    end)
    Frame.InputEnded:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 then drag = false end
    end)
    game:GetService("UserInputService").InputChanged:Connect(function(i)
        if drag and i.UserInputType == Enum.UserInputType.MouseMovement then
            local delta = i.Position - dragStart
            Frame.Position = UDim2.new(
                startPos.X.Scale, startPos.X.Offset + delta.X,
                startPos.Y.Scale, startPos.Y.Offset + delta.Y
            )
        end
    end)

    f.BackgroundColor3 = Color3.fromRGB(18, 18, 24)

    return f
end

local lastCount = 0
RunService.Heartbeat:Connect(function()
    if #logs == lastCount then return end
    lastCount = #logs

    for _, c in pairs(Scroll:GetChildren()) do
        if c:IsA("Frame") then
            c._lbl.Text = ""
            c.Parent = nil
            table.insert(rowPool, c)
        end
    end

    local order = 0
    for _, entry in ipairs(logs) do
        local isOut = entry.tag:find("OUT")
        local goal  = (type(entry.args[1]) == "table" and entry.args[1].Goal) and entry.args[1].Goal or nil

        local lines = {
            string.format("%s [%s] %s  ·  %s%s",
                entry.time, entry.tag, entry.remote, entry.source,
                goal and ("  Goal: "..goal) or ""),
            "    "..formatArgs(entry.args),
        }
        local text = table.concat(lines, "\n")

        local row = getRow()
        row.LayoutOrder = order
        order += 1

        local lineCount = select(2, text:gsub("\n", "\n")) + 1
        row.Size = UDim2.new(1, 0, 0, 14 * lineCount + 8)
        row.BackgroundColor3 = isOut
            and Color3.fromRGB(20, 28, 20)
            or  Color3.fromRGB(20, 20, 32)

        row._lbl.Text = text
        row._lbl.TextColor3 = isOut
            and Color3.fromRGB(150, 255, 160)
            or  Color3.fromRGB(140, 180, 255)

        row.Parent = Scroll
    end

    Scroll.CanvasSize = UDim2.new(0, 0, 0, Layout.AbsoluteContentSize.Y + 4)
end)
