local function safeGet(t, k)
    local ok, v = pcall(function() return t[k] end)
    return ok and v or nil
end

local function serialize(v, depth)
    depth = depth or 0
    if depth > 3 then return "..." end
    local t = type(v)
    if t == "string"  then return '"' .. tostring(v):sub(1, 80) .. '"' end
    if t == "number"  then return tostring(v) end
    if t == "boolean" then return tostring(v) end
    if t == "nil"     then return "nil" end
    if t == "table"   then
        local parts = {}
        local i = 0
        for k2, v2 in next, v do
            i = i + 1
            if i > 8 then table.insert(parts, "..."); break end
            table.insert(parts, tostring(k2) .. "=" .. serialize(v2, depth + 1))
        end
        return "{" .. table.concat(parts, ", ") .. "}"
    end
    if t == "userdata" or t == "Instance" then
        local ok2, name = pcall(tostring, v)
        return ok2 and name or "userdata"
    end
    return t
end

local function serializeArgs(args)
    local parts = {}
    for i, v in ipairs(args) do
        parts[i] = "[" .. i .. "] " .. serialize(v)
    end
    return table.concat(parts, "  |  ")
end

local function tryGetSource(remote)
    local src = nil
    pcall(function()
        for _, c in ipairs(remote:GetDescendants()) do
            local cls = safeGet(c, "ClassName")
            if cls == "LocalScript" or cls == "Script" or cls == "ModuleScript" then
                if decompile then
                    local ok, code = pcall(decompile, c)
                    if ok and type(code) == "string" and #code > 0 then
                        src = "-- [decompile] " .. c:GetFullName() .. "\n" .. code
                        return
                    end
                end
                src = "-- [script, no decompile] " .. c:GetFullName()
                return
            end
        end
    end)
    if src then return src end
    pcall(function()
        local meta = getrawmetatable and getrawmetatable(remote)
        if not meta then return end
        for _, k in ipairs({"__namecall","__index","__newindex","__call"}) do
            local raw = rawget(meta, k)
            if raw and type(raw) == "function" then
                local info = debug.getinfo and debug.getinfo(raw, "S")
                if info and info.short_src and info.short_src ~= "[C]" then
                    src = "-- [hook: " .. k .. "]\n-- src: " .. info.short_src
                    if info.linedefined then src = src .. "\n-- line: " .. info.linedefined end
                    if decompile then
                        local ok, code = pcall(decompile, raw)
                        if ok and code and #code > 0 then src = src .. "\n\n" .. code end
                    end
                    return
                end
            end
        end
    end)
    if src then return src end
    return "-- Нет исходника\n-- Remote: " .. remote:GetFullName()
end

local SCAN_MODES = {
    Fast  = { depth = 2,  checkMeta = false, metaKeys = {},                                                                label = "FAST",  color = Color3.fromRGB(40,160,40)   },
    Quick = { depth = 4,  checkMeta = true,  metaKeys = {"__namecall","__index"},                                          label = "QUICK", color = Color3.fromRGB(30,90,200)   },
    Scan  = { depth = 6,  checkMeta = true,  metaKeys = {"__namecall","__index","__newindex","__call","__len"},             label = "SCAN",  color = Color3.fromRGB(160,120,20)  },
    Deep  = { depth = 10, checkMeta = true,  metaKeys = {"__namecall","__index","__newindex","__call","__len","__concat","__tostring","__eq","__lt","__le","__add","__sub","__mul","__div","__mod","__pow","__unm","__metatable"}, label = "DEEP", color = Color3.fromRGB(180,30,30) },
}

local Scanner = {}
function Scanner.run(modeName, onItem, onDone)
    local cfg = SCAN_MODES[modeName]
    if not cfg then onDone(0) return end
    local found, visited = 0, {}
    local queue = {{ inst = game, depth = 0 }}
    local function processChunk()
        local CHUNK = 300
        local processed = 0
        while #queue > 0 and processed < CHUNK do
            local item   = table.remove(queue, 1)
            local inst   = item.inst
            local depth  = item.depth
            processed    = processed + 1
            if depth > cfg.depth then goto next end
            local id = tostring(inst)
            if visited[id] then goto next end
            visited[id] = true
            local cls = safeGet(inst, "ClassName")
            if cls == "RemoteEvent" or cls == "RemoteFunction" then
                found = found + 1
                local entry = { ref = inst, name = inst:GetFullName(), class = cls, hooks = {}, hooked = false }
                if cfg.checkMeta then
                    pcall(function()
                        local meta = getrawmetatable and getrawmetatable(inst)
                        if not meta then return end
                        for _, key in ipairs(cfg.metaKeys) do
                            local raw = rawget(meta, key)
                            if raw and type(raw) == "function" then
                                local src = "?"
                                pcall(function() local i = debug.getinfo(raw,"S"); src = i and i.short_src or "?" end)
                                if src ~= "[C]" and src ~= "?" then
                                    entry.hooked = true
                                    table.insert(entry.hooks, { key = key, source = src, fn = raw })
                                end
                            end
                        end
                    end)
                end
                onItem(entry)
            end
            if depth < cfg.depth then
                pcall(function()
                    for _, child in ipairs(inst:GetChildren()) do
                        table.insert(queue, { inst = child, depth = depth + 1 })
                    end
                end)
            end
            ::next::
        end
        if #queue > 0 then task.defer(processChunk)
        else onDone(found) end
    end
    task.defer(processChunk)
end

local function hookRemote(remote, onFired)
    -- Хукаем через __namecall метаметод
    -- Проще и надёжнее — hookfunction на сам метод
    local cls = safeGet(remote, "ClassName")
    if cls == "RemoteEvent" then
        -- hookfunction(remote.FireServer, ...) не работает напрямую в Roblox
        -- Используем getrawmetatable + __namecall перехват на конкретный объект
        local ok = pcall(function()
            local meta = getrawmetatable(remote)
            if not meta then return end
            if setreadonly then pcall(setreadonly, meta, false) end
            local oldNC = rawget(meta, "__namecall")
            rawset(meta, "__namecall", newcclosure and newcclosure(function(self, ...)
                local method = getnamecallmethod and getnamecallmethod() or ""
                if self == remote and (method == "FireServer" or method == "InvokeServer") then
                    local args = {...}
                    onFired(method, args)
                end
                if oldNC then return oldNC(self, ...) end
                return (self[method])(self, ...)
            end) or function(self, ...)
                local method = getnamecallmethod and getnamecallmethod() or ""
                if self == remote and (method == "FireServer" or method == "InvokeServer") then
                    onFired(method, {...})
                end
                if oldNC then return oldNC(self, ...) end
            end)
            if setreadonly then pcall(setreadonly, meta, true) end
        end)
        _ = ok
    elseif cls == "RemoteFunction" then
        pcall(function()
            local meta = getrawmetatable(remote)
            if not meta then return end
            if setreadonly then pcall(setreadonly, meta, false) end
            local oldNC = rawget(meta, "__namecall")
            rawset(meta, "__namecall", newcclosure and newcclosure(function(self, ...)
                local method = getnamecallmethod and getnamecallmethod() or ""
                if self == remote and method == "InvokeServer" then
                    onFired(method, {...})
                end
                if oldNC then return oldNC(self, ...) end
            end) or function(self, ...)
                local method = getnamecallmethod and getnamecallmethod() or ""
                if self == remote and method == "InvokeServer" then
                    onFired(method, {...})
                end
                if oldNC then return oldNC(self, ...) end
            end)
            if setreadonly then pcall(setreadonly, meta, true) end
        end)
    end
end

-- Глобальный __namecall хук (если есть hookmetamethod)
local globalHookRestorer = nil

local function startGlobalSpy(filter, onFired)
    -- filter: nil = все ремоуты, "Communicate" = только с этим именем
    if hookmetamethod then
        -- Самый мощный способ
        local meta = getrawmetatable(game)
        globalHookRestorer = hookmetamethod(game, "__namecall", newcclosure and newcclosure(function(self, ...)
            local method = getnamecallmethod and getnamecallmethod() or ""
            if method == "FireServer" or method == "InvokeServer" then
                local cls = safeGet(self, "ClassName")
                if cls == "RemoteEvent" or cls == "RemoteFunction" then
                    local name = tostring(self.Name or "?")
                    if not filter or name:find(filter) then
                        onFired(self, method, {...})
                    end
                end
            end
            return globalHookRestorer(self, ...)
        end) or function(self, ...)
            local method = getnamecallmethod and getnamecallmethod() or ""
            if method == "FireServer" or method == "InvokeServer" then
                local cls = safeGet(self, "ClassName")
                if cls == "RemoteEvent" or cls == "RemoteFunction" then
                    local name = tostring(self.Name or "?")
                    if not filter or name:find(filter) then
                        onFired(self, method, {...})
                    end
                end
            end
            return globalHookRestorer(self, ...)
        end)
        return true
    end
    return false
end

local function stopGlobalSpy()
    globalHookRestorer = nil
end

                if setreadonly then pcall(setreadonly,meta,false) end
                for _,h in ipairs(entry.hooks) do pcall(rawset,meta,h.key,nil) end
                if setreadonly then pcall(setreadonly,meta,true) end
            end)
            entry.hooked=false; entry.hooks={}
            Row.BackgroundColor3=Color3.fromRGB(16,20,16)
            StatusLbl.Text="Нукнули: " .. entry.name
            StatusLbl.TextColor3=Color3.fromRGB(80,220,120)
        end)

        table.insert(remoteEntries, entry)
    end

    local scanning = false

    local function runScan(modeName)
        if scanning then return end
        scanning = true
        -- Очистить
        for _, c in ipairs(LS:GetChildren()) do
            if c:IsA("Frame") then c:Destroy() end
        end
        remoteEntries = {}
        local cfg = SCAN_MODES[modeName]
        StatusLbl.Text      = "[ " .. cfg.label .. " ] Сканирую..."
        StatusLbl.TextColor3 = cfg.color
        CountLbl.Text       = ""
        local t0    = os.clock()
        local total = 0
        Scanner.run(modeName, function(entry)
            total = total + 1
            addRow(entry)
            CountLbl.Text = tostring(total)
        end, function(count)
            local el = string.format("%.2f", os.clock()-t0)
            StatusLbl.Text       = string.format("[ %s ] %d ремоутов — %sс", cfg.label, count, el)
            StatusLbl.TextColor3 = Color3.fromRGB(80,220,120)
            CountLbl.Text        = tostring(count) .. " rem"
            scanning = false
        end)
    end

    for _, m in ipairs(MODES) do
        local b = btn(BtnRow,{Size=UDim2.new(0,115,0,34),BackgroundColor3=m.col,Text=m.text,TextSize=12})
        local mode = m.mode
        b.MouseButton1Click:Connect(function() runScan(mode) end)
    end

    ClearBtn.MouseButton1Click:Connect(function()
        for _, c in ipairs(LS:GetChildren()) do if c:IsA("Frame") then c:Destroy() end end

    end)

                    if not meta then return end
                    if setreadonly then pcall(setreadonly,meta,false) end
                    for _,h in ipairs(e.hooks) do pcall(rawset,meta,h.key,nil) end
                    if setreadonly then pcall(setreadonly,meta,true) end
                end)
                e.hooked=false; e.hooks={}; n=n+1
            end
        end
        StatusLbl.Text="Нукнуто " .. n .. " хуков"; StatusLbl.TextColor3=Color3.fromRGB(255,80,80)
    end)
    CopyBtn.MouseButton1Click:Connect(function()
        local lines={}
        for _,e in ipairs(remoteEntries) do
            table.insert(lines, (e.hooked and "[HOOK] " or "[ok]   ") .. e.name)
            for _,h in ipairs(e.hooks) do table.insert(lines,"  -> "..h.key.." @ "..h.source) end
        end
        local txt=table.concat(lines,"\n")
        if setclipboard then pcall(setclipboard,txt) end
        StatusLbl.Text="Скопировано"; StatusLbl.TextColor3=Color3.fromRGB(100,180,255)
    end)

    -- Авто-запуск
    task.delay(0.5, function() runScan("Quick") end)
end
